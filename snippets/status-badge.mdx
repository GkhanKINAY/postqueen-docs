{/*
  A status badge for a feature that is not out yet (owner, 2026-09-24): Make,
  Zapier, the Chrome extension and clipping. Networks carry no status badge.
    soon       gray   "Soon"
    available  green  "Available"

  import { ChannelStatus } from '/snippets/status-badge.mdx';
  <ChannelStatus status="soon">Not available on PostQueen yet.</ChannelStatus>

  ChannelBadge is the same badge on its own, with no note and no margin, for a
  table cell or a card.
*/}

export const ChannelStatus = ({ status, children }) => {
  const look = {
    available: { color: 'green', label: 'Available' },
    soon: { color: 'gray', label: 'Soon' },
  }[status] || { color: 'gray', label: status };
  return (
    <div className="pq-status not-prose" data-status={status}>
      <Badge color={look.color} shape="pill" size="md">{look.label}</Badge>
      {children ? <span className="pq-status-note">{children}</span> : null}
    </div>
  );
};

export const ChannelBadge = ({ status }) => {
  const look = {
    available: { color: 'green', label: 'Available' },
    soon: { color: 'gray', label: 'Soon' },
  }[status] || { color: 'gray', label: status };
  return <Badge color={look.color} shape="pill" size="sm">{look.label}</Badge>;
};
