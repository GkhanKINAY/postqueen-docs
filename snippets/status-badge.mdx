{/*
  Channel status, one line under a channel page's title. The status must equal
  facts/facts.json for that network (npm run check:facts compares them):
    available  = works           green   "Available"
    in-review  = review_gated    yellow  "In review", the note is the limit
    soon       = no_keys, cannot_connect   gray  "Soon"
  The note (children) is the network's statusNote from facts.json, word for word.

  import { ChannelStatus } from '/snippets/status-badge.mdx';
  <ChannelStatus status="in-review">Until Meta approves PostQueen's app, ...</ChannelStatus>

  ChannelBadge is the same badge on its own, with no note and no margin, for
  a table cell or a card on the index pages (channels/overview,
  public-api/providers/overview):
  <ChannelBadge status="soon" />
*/}

export const ChannelStatus = ({ status, children }) => {
  const look = {
    available: { color: 'green', label: 'Available' },
    'in-review': { color: 'yellow', label: 'In review' },
    soon: { color: 'gray', label: 'Soon' },
  }[status] || { color: 'gray', label: status };
  return (
    <div className="pq-status not-prose" data-status={status}>
      <Badge color={look.color} shape="pill" size="md">{look.label}</Badge>
      {children ? <span className="pq-status-note">{children}</span> : null}
    </div>
  );
};

export const ChannelBadge = ({ status }) => {
  const look = {
    available: { color: 'green', label: 'Available' },
    'in-review': { color: 'yellow', label: 'In review' },
    soon: { color: 'gray', label: 'Soon' },
  }[status] || { color: 'gray', label: status };
  return <Badge color={look.color} shape="pill" size="sm">{look.label}</Badge>;
};
